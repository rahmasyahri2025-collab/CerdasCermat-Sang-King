        function playCorrectSound() {
            try {
                if(Tone.context.state !== 'running') Tone.start();
                const synth = new Tone.PolySynth(Tone.Synth).toDestination();
                const now = Tone.now();
                synth.volume.value = -5;
                synth.triggerAttackRelease(["C5", "E5", "G5"], "8n", now);
                synth.triggerAttackRelease(["E5", "G5", "C6"], "4n", now + 0.15);
            } catch(e) { console.log("Audio not initialized"); }
        }

        function playWrongSound() {
            try {
                if(Tone.context.state !== 'running') Tone.start();
                const synth = new Tone.Synth({ oscillator: { type: "sawtooth" } }).toDestination();
                const now = Tone.now();
                synth.volume.value = -5;
                synth.triggerAttackRelease("D2", "8n", now);
                synth.triggerAttackRelease("C2", "2n", now + 0.2);
            } catch(e) { console.log("Audio not initialized"); }
        }

        function showError(title, message) {
            document.getElementById('error-title').innerText = title;
            document.getElementById('error-message').innerText = message;
            document.getElementById('error-modal').classList.remove('hide');
            console.error(title, message);
        }

        async function initFirebase() {
            try {
                if (!appId || !firebaseConfig || Object.keys(firebaseConfig).length === 0) {
                    throw new Error("Konfigurasi Firebase tidak ditemukan di environment ini.");
                }
                app = initializeApp(firebaseConfig);
                db = getFirestore(app);
                auth = getAuth(app);

                const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
                if (token) {
                    await signInWithCustomToken(auth, token);
                } else {
                    await signInAnonymously(auth);
                }
                
                onAuthStateChanged(auth, (user) => {
                    if (user) {
                        userId = user.uid;
                        setupDatabaseAndListeners();
                    }
                });
            } catch (error) {
                showError("Gagal Inisialisasi", error.message);
                screens.loading.classList.add('hide');
            }
        }

        async function setupDatabaseAndListeners() {
            try {
                const groupsRef = collection(db, 'artifacts', appId, 'public', 'data', 'groups');
                const groupsSnap = await getDocs(groupsRef);
                
                if (groupsSnap.empty) {
                    const batch = writeBatch(db);
                    for(let i=0; i<=8; i++) {
                        const gRef = doc(db, 'artifacts', appId, 'public', 'data', 'groups', `group${i}`);
                        batch.set(gRef, { name: i === 0 ? 'Guru' : `Kelompok ${i}`, claimedBy: null, score: 0 });
                    }
                    const stateRef = doc(db, 'artifacts', appId, 'public', 'data', 'gamestate', 'state');
                    batch.set(stateRef, { status: 'lobby', currentQuestion: 0, startTime: 0 });
                    await batch.commit();
                }

                const stateRef = doc(db, 'artifacts', appId, 'public', 'data', 'gamestate', 'state');
                onSnapshot(stateRef, (docSnap) => {
                    if(docSnap.exists()) {
                        currentGameState = docSnap.data();
                        handleStateChange();
                    }
                }, (error) => {
                    showError("Sync Error", error.message);
                });

                onSnapshot(groupsRef, (querySnap) => {
                    allGroups = {};
                    querySnap.forEach(doc => {
                        allGroups[doc.id] = doc.data();
                        // CRITICAL FIX: Ensure Teacher state persists across reloads!
                        if(doc.data().claimedBy === userId) {
                            myGroupId = doc.id;
                            isTeacher = (myGroupId === 'group0');
                        }
                    });
                    
                    if (currentGameState.status === 'lobby') {
                        if (!myGroupId) {
                            renderLoginScreen();
                        } else {
                            renderLobbyScreen();
                        }
                    } else if (currentGameState.status === 'playing') {
                        if (!myGroupId) {
                            alert("Permainan sudah dimulai. Anda menjadi penonton.");
                            renderLobbyScreen(); 
                        }
                    }
                }, (error) => {
                    showError("Groups Sync Error", error.message);
                });

            } catch (err) {
                showError("Setup DB Error", err.message);
            }
        }

        function switchScreen(screenId) {
            Object.values(screens).forEach(s => s.classList.add('hide'));
            screens[screenId].classList.remove('hide');

            // Header Banner hanya di Login & Lobby
            if(screenId === 'login' || screenId === 'lobby') {
                headerBanner.classList.remove('hide');
            } else {
                headerBanner.classList.add('hide');
            }
        }

        function handleStateChange() {
            const status = currentGameState.status;
            
            if (status === 'lobby') {
                if (myGroupId) renderLobbyScreen();
                else renderLoginScreen();
            } else if (status === 'playing') {
                startQuestionUI();
            } else if (status === 'evaluation') {
                renderEvaluationScreen();
            } else if (status === 'finished') {
                renderFinalScreen();
            }
        }

        function startTimer(durationSeconds) {
            clearInterval(timerInterval);
            const timerText = document.getElementById('timer-text');
            const timerBar = document.getElementById('timer-bar');
            
            const targetTime = currentGameState.startTime + (durationSeconds * 1000);
            
            timerInterval = setInterval(() => {
                const now = Date.now();
                const remainingMs = targetTime - now;
                const remainingSec = Math.max(0, Math.ceil(remainingMs / 1000));
                
                timerText.innerText = remainingSec;
                
                const percent = Math.max(0, (remainingMs / (durationSeconds * 1000)) * 100);
                timerBar.style.width = `${percent}%`;
                
                if(percent < 30) timerBar.className = "timer-bar timer-warning";
                if(percent < 10) timerBar.className = "timer-bar timer-danger";
                if(percent >= 30) timerBar.className = "timer-bar";

                if (remainingMs <= 0) {
                    clearInterval(timerInterval);
                    timerText.innerText = "0";
                    forceEndQuestion();
                }
            }, 100);
        }

        async function submitAnswer(optIndex) {
            if (!userId || !myGroupId) return;
            const qIndex = currentGameState.currentQuestion;
            
            if (currentQuestionAnswers[myGroupId]) return;

            document.getElementById('answer-wait-overlay').classList.remove('hide');

            const answerDocRef = doc(db, 'artifacts', appId, 'public', 'data', `answers_q${qIndex}`, myGroupId);
            try {
                await setDoc(answerDocRef, {
                    answerIndex: optIndex,
                    timestamp: Date.now()
                });
            } catch (err) {
                console.error("Gagal mengirim jawaban", err);
                document.getElementById('answer-wait-overlay').classList.add('hide');
            }
        }

        function checkAllAnswered(totalTimeSec) {
            let playingGroupsCount = 0;
            for(let key in allGroups) {
                if (key !== 'group0' && allGroups[key].claimedBy) playingGroupsCount++;
            }
            
            const answersCount = Object.keys(currentQuestionAnswers).length;
            
            if (playingGroupsCount > 0 && answersCount >= playingGroupsCount) {
                clearInterval(timerInterval);
                forceEndQuestion();
            }
        }

        async function forceEndQuestion() {

                const stateRef = doc(db, 'artifacts', appId, 'public', 'data', 'gamestate', 'state');
                await updateDoc(stateRef, { status: 'evaluation' });
            }
        }

        async function renderEvaluationScreen() {
            switchScreen('evaluation');
            if(localAnswersUnsubscribe) { localAnswersUnsubscribe(); localAnswersUnsubscribe = null; }
            clearInterval(timerInterval);

            const qIndex = currentGameState.currentQuestion;
            const qData = QUESTIONS[qIndex];
            
            document.getElementById('eval-correct-answer').innerText = `${String.fromCharCode(65 + qData.correct)}. ${qData.options[qData.correct]}`;
            
            let answersArray = [];
            for (let gId in currentQuestionAnswers) {
                if(gId === 'group0') continue;
                answersArray.push({
                    groupId: gId,
                    answerIndex: currentQuestionAnswers[gId].answerIndex,
                    timestamp: currentQuestionAnswers[gId].timestamp
                });
            }
            
            answersArray.sort((a, b) => a.timestamp - b.timestamp);
            
            let fastestFound = false;
            let pointsEarned = 0;
            let isCorrect = false;
            
            for (let ans of answersArray) {
                if (ans.answerIndex === qData.correct) {
                    if (!fastestFound) {
                        ans.points = 100;
                        fastestFound = true;
                    } else {
                        ans.points = 90;
                    }
                    if(ans.groupId === myGroupId) isCorrect = true;
                } else {
                    ans.points = 0;
                }
                if (ans.groupId === myGroupId) pointsEarned = ans.points;
            }
            
            if (isTeacher) {
                const batch = writeBatch(db);
                let analysisData = {};
                
                answersArray.forEach(ans => {
                    if (ans.points > 0) {
                        const gRef = doc(db, 'artifacts', appId, 'public', 'data', 'groups', ans.groupId);
                        batch.update(gRef, { score: increment(ans.points) });
                    }
                    analysisData[ans.groupId] = {
                        answer: ans.answerIndex,
                        points: ans.points,
                        correct: (ans.answerIndex === qData.correct)
                    };
                });
                
                const analysisRef = doc(db, 'artifacts', appId, 'public', 'data', 'analysis', `q${qIndex}`);
                batch.set(analysisRef, analysisData);
                
                try {
                    await batch.commit();
                } catch(e) { console.error("Error updating scores", e); }
            }

            const teacherView = document.getElementById('teacher-eval-view');
            const studentView = document.getElementById('student-eval-view');
            const evalBadge = document.getElementById('eval-badge');
            const waitMsg = document.getElementById('eval-waiting-msg');

            if (isTeacher) {
                teacherView.classList.remove('hide');
                studentView.classList.add('hide');
                waitMsg.classList.add('hide');
                setTimeout(renderLocalLeaderboard, 1000);
                
                const btnNext = document.getElementById('btn-next-question');
                btnNext.onclick = nextQuestion;
                if(qIndex === QUESTIONS.length - 1) {
                    btnNext.innerText = "Selesaikan Kuis & Lihat Hasil ➔";
                    btnNext.className = "mt-6 bg-gradient-to-r from-yellow-500 to-orange-500 text-white font-black py-4 px-8 rounded-2xl shadow-xl hover:shadow-2xl transition-all w-full text-xl transform hover:-translate-y-1";
                }
            } else {
                teacherView.classList.add('hide');
                studentView.classList.remove('hide');
                waitMsg.classList.remove('hide');

                const didAnswer = answersArray.find(a => a.groupId === myGroupId);
                
                if (isCorrect) {
                    playCorrectSound();

        async function nextQuestion() {
            if (!isTeacher) return;
            const stateRef = doc(db, 'artifacts', appId, 'public', 'data', 'gamestate', 'state');
            
            if (currentGameState.currentQuestion >= QUESTIONS.length - 1) {
                await updateDoc(stateRef, { status: 'finished' });
            } else {
                await updateDoc(stateRef, {

        async function loadAnalysisData() {
            const btn = document.getElementById('btn-load-analysis');
            const loading = document.getElementById('analysis-loading');
            const table = document.getElementById('analysis-table');
            
            btn.disabled = true;
            btn.innerText = "Memuat...";
            loading.classList.remove('hide');
            table.classList.add('hide');

            try {
                const analysisSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'analysis'));
                let recapData = {}; 
                
                analysisSnap.forEach(doc => {
                    recapData[doc.id] = doc.data();
                });

                const activeGroups = Object.keys(allGroups)
                    .filter(k => k !== 'group0' && allGroups[k].claimedBy !== null)
                    .sort((a,b) => parseInt(a.replace('group','')) - parseInt(b.replace('group','')));

                const totalQuestions = currentGameState.currentQuestion + (currentGameState.status === 'finished' ? 1 : 0);

                let html = '<thead><tr><th class="w-16">Soal</th>';
                activeGroups.forEach(gId => {
                    html += `<th>Klp ${gId.replace('group', '')}</th>`;
                });
                html += '</tr></thead><tbody>';

                for (let i = 0; i < totalQuestions; i++) {
                    const qKey = `q${i}`;
                    const qResult = recapData[qKey] || {};
                    
                    html += `<tr><td class="font-black bg-gray-100 border-b-2 text-gray-700">${i+1}</td>`;
                    
                    activeGroups.forEach(gId => {
                        const res = qResult[gId];
                        if (!res) {
                            html += `<td class="cell-empty font-bold text-gray-400">-</td>`;
                        } else if (res.correct) {
                            html += `<td class="cell-correct">✓<br><span class="text-[10px] text-green-600 font-black">${res.points}p</span></td>`;
                        } else {
                            html += `<td class="cell-wrong font-black">✗</td>`;
                        }
                    });
                    html += '</tr>';
                }
                html += '</tbody>';

                table.innerHTML = html;
                loading.classList.add('hide');
                table.classList.remove('hide');
                btn.innerText = "Refresh Data";
                btn.disabled = false;

            } catch (error) {
                console.error("Error loading analysis", error);
                loading.innerText = "Gagal memuat data.";
                btn.innerText = "Coba Lagi";
                btn.disabled = false;
            }
        }
